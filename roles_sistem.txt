(P-ACME)_ACME   |   (P-App)_App    |  (P-Banca)_Banca   |  (P-Fleet)_Fleet

P-ACME  ::= (
    (
        list_disp_req@ACME ; ~list_disp_res@ACME ; QR_code_req@ACME ; ~QR_code_res@ACME ;
        ~blocco_cauzione@ACME ; token@ACME ;

        ~notifica_sblocco@ACME ;  ( ~status@ACME -; res@ACME )* ;
        riconsenga@ACME ; ~notifica_blocco@ACME ; 
        ~addebito@ACME; token@ACME ;
        ~sblocco_cauzione@ACME; token@ACME ; 
    )
    +
    (
        bloccare_veicolo@ACME ; 
        ~blocco_cauzione@ACME ; token@ACME ;
        ( annullare@ACME ; ~sblocco_cauzione@ACME; token@ACME )
        +
        ( ~convertire_addebito@ACME; token@ACME )
        +
        ( 
            ~notifica_sblocco@ACME ; sbloccare_veicolo@ACME ; ( ~status@ACME -; res@ACME )* ;
            riconsenga@ACME ; ~notifica_blocco@ACME ; 
            ~addebito@ACME; token@ACME ;
            ~sblocco_cauzione@ACME; token@ACME ; 
        )
    )
)*

P-App  ::= (
    (
        ~list_disp_req@ACME ; list_disp_res@ACME ; 
        ~QR_code_req@ACME ; QR_code_res@ACME ;
        ~riconsenga@ACME ; 
    )
    +
    (
        ~bloccare_veicolo@ACME ; 
        ( ~annullare@ACME  )
        +
        ( ~sbloccare_veicolo@ACME ; ~riconsenga@ACME ; )
    )
)*

P-Banca  ::= (
    blocco_cauzione@ACME ; ~token@ACME ;
    ( sblocco_cauzione@ACME ; ~token@ACME ; )
    +
    ( convertire_addebito@ACME; ~token@ACME )
    +
    ( addebito@ACME ; ~token@ACME ; sblocco_cauzione@ACME ; ~token@ACME ; )
)*

P-Fleet  ::= (
    notifica_sblocco@ACME ; 
    ( status@ACME ; ~res@ACME )* ;
    notifica_blocco@ACME ;
)*
