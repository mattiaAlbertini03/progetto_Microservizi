 (ACME-P)_ACME | (App-P)_App | (Stazione-P)_Stazione | (Banca-P)_Banca | (Fleet-P)_Fleet

Fleet-P ::= (
	notifica_sblocco@ACME ; ~inizio@ACME ;  
	( status_req@ ACME; ~status_res@ACME )* ;
	notifica_blocco@ACME; ~fine:@ ACME;  
)*

Banca-P ::= (
	( addebbito@ACME ;   ~token@ACME ; )
	+
	(
	Cauzione@ACME ;   ~token@ACME ;  
    	addebito@ACME ;   ~token@ACME ;  
	sblocca_cauzione@ACME ;   ~token@ACME ;  
	)
)*

Stazione-P ::= (
	sblocco@ACME; ~esito@ACME ; 
	blocco@ACME;  ~esito@ACME ;	
)*

App-P ::= (
	( 	
		~mappa_req@ ACME;  mappa_ris@ACME ; 
		~QR_ritiro@ ACME;  ris_utente@ACME;
		~Riconsegna@ ACME; ris_finale@ACME; 
	)
	+
	(	
	~prenotazione@ ACME; ris_pren@ACME; 
     	( ~Anulla@ ACME; Anulla_ris@ACME )
	+
	(  ~QR_ritiro@ ACME;  ris_utente@ACME; ~Riconsegna@ ACME; ris_finale@ACME; )
	+
	( 1 )
    )	
)*


ACME ::= (
	( 
		mappa_req@App ;  ~mappa_ris@App ;  
		QR_ritiro@App ;  
		~sblocco@Stazione ; esito@Stazione ; 
		~Cauzione@Banca ; token@Banca ;
		~notifica_sblocco@Fleet ; inizio@Fleet ; 
		~ris_utente@App; 
		(~status_req@Fleet ; status_res@Fleet ; )* ;
		Riconsegna@App ; ~blocco@Stazione ; esito@Stazione ; 
		~notifica_blocco@Fleet ; fine@Fleet ;  
	    ~addebito@Banca ; token@Banca ; 
		~sblocca_cauzione@Banca ; token@Banca ;
		~ris_finale@App; 
	)
	+
	(
    	prenotazione@App ;  ~ris_pre@App;
		(
			Annulla@App ; ~annulla_ris@App;
		    	(~addebbito@Banca ; token@Banca ; ) + (1) ;
		)
	    +
    		(~addebbito@Banca ; token@Banca ;)
	    +
		(
			ritiro@App ; 
		~sblocco@Stazione ; esito@Stazione ; 
		~Cauzione@Banca ; token@Banca ;
		~notifica_sblocco@Fleet ; inizio@Fleet ; 
		~ris_utente@App; 
		(~status_req@Fleet ; status_res@Fleet ; )* ;
		Riconsegna@App ; ~blocco@Stazione ; esito@Stazione ; 
		~notifica_blocco@Fleet ; fine@Fleet ;  
	    ~addebito@Banca ; token@Banca ; 
		~sblocca_cauzione@Banca ; token@Banca ;
		~ris_finale@App; 
	    	
		)
	)
)*
