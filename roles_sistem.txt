(P-ACME)_ACME   |   (P-App)_App    |  (P-Banca)_Banca   |  (P-Fleet)_Fleet

P-ACME  ::= (
    (
        list_disp_req@App ; ~list_disp_res@App ; QR_code_req@App ; ~QR_code_res@App ;
        ~blocco_cauzione@Banca ; token@Banca ;

        ~notifica_sblocco@Fleet ;  ( ~status@Fleet ; res@Fleet )* ;
        riconsenga@App ; ~notifica_blocco@Fleet ; 
        ~addebito@Banca; token@Banca ;
        ~sblocco_cauzione@Banca; token@Banca ; 
    )
    +
    (
        bloccare_veicolo@App ; 
        ~blocco_cauzione@Banca ; token@Banca ;
        ( annullare@ACME ; ~sblocco_cauzione@Banca; token@Banca )
        +
        ( ~convertire_addebito@Banca; token@Banca )
        +
        ( 
            ~notifica_sblocco@Fleet ; sbloccare_veicolo@App ;  ( ~status@Fleet ; res@Fleet )* ;
            riconsenga@App ; ~notifica_blocco@Fleet ; 
            ~addebito@Banca; token@Banca ;
            ~sblocco_cauzione@Banca; token@Banca ; 
        )
    )
)*

P-App  ::= (
    (
        ~list_disp_req@ACME ; list_disp_res@ACME ; 
        ~QR_code_req@ACME ; QR_code_res@ACME ;
        ~riconsenga@ACME ; 
    )
    +
    (
        ~bloccare_veicolo@ACME ; 
        ( ~annullare@ACME  )
        +
        ( ~sbloccare_veicolo@ACME ; ~riconsenga@ACME ; )
    )
)*

P-Banca  ::= (
    blocco_cauzione@ACME ; ~token@ACME ;
    ( sblocco_cauzione@ACME ; ~token@ACME ; )
    +
    ( convertire_addebito@ACME; ~token@ACME )
    +
    ( addebito@ACME ; ~token@ACME ; sblocco_cauzione@ACME ; ~token@ACME ; )
)*

P-Fleet  ::= (
    notifica_sblocco@ACME ; 
    ( status@ACME ; ~res@ACME )* ;
    notifica_blocco@ACME ;
)*
